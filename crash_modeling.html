crashes <- read_rds("data/nyc_collisions_clean.rds")
crashes_analysis <- crashes |>
  mutate(
    # Outcomes
    injury_crash = number_of_persons_injured > 0,
    fatal_crash = number_of_persons_killed > 0,

    # Time
    year = year(crash_date),
    month = month(crash_date, label = TRUE),
    day_of_week = wday(crash_date, label = TRUE),

    # Keep original crash_time unchanged
    hour = as.integer(str_sub(crash_time, 1, 2)),

    weekend = day_of_week %in% c("Sat", "Sun"),
    night = hour < 6 | hour >= 20,
    rush_hour = hour %in% c(7, 8, 9, 16, 17, 18)
  )
crashes_analysis <- crashes_analysis |>
  mutate(
